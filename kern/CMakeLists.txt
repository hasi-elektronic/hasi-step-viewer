# Nativer Kern für den Hasi STEP-Betrachter. OCCT wird dynamisch gelinkt (LGPL-2.1 mit OCCT-Ausnahme).
cmake_minimum_required(VERSION 3.16)
project(hasi_kern CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(OpenCASCADE CONFIG REQUIRED)
add_executable(hasi-kern hasi_kern.cpp)
target_include_directories(hasi-kern PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_link_libraries(hasi-kern PRIVATE TKDESTEP TKDE TKXSBase TKMesh TKBRep TKTopAlgo TKShHealing TKGeomBase TKG3d TKG2d TKMath TKernel)
find_package(Threads REQUIRED)
target_link_libraries(hasi-kern PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(hasi-kern PRIVATE /utf-8 /EHsc /bigobj)
endif()
if(APPLE)
  # In der App liegen die Bibliotheken unter Contents/Resources/kern
  set_target_properties(hasi-kern PROPERTIES BUILD_RPATH "${OpenCASCADE_LIBRARY_DIR}" INSTALL_RPATH "@executable_path/../Resources/kern")
  target_link_options(hasi-kern PRIVATE "-Wl,-rpath,@executable_path/../Resources/kern")
elseif(UNIX)
  set_target_properties(hasi-kern PROPERTIES BUILD_RPATH "${OpenCASCADE_LIBRARY_DIR}")
endif()
